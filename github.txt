  SECTION 3 — ADD REMOTE ORIGIN
----------------------------------------------------------------

# Add the remote repository (your GitHub repo URL)

# ✅ Expected output: (no output = success)


----------------------------------------------------------------
  SECTION 4 — CHECK / VERIFY REMOTE
----------------------------------------------------------------

# List all remotes (shows name and URL)
git remote -v

# ✅ Expected output:
# origin  https://github.com/aamir490/Use-case--9-NovaMind-AI-Text-to-Speech-Converter-with-amazone-polly.git (fetch)
# origin  https://github.com/aamir490/Use-case--9-NovaMind-AI-Text-to-Speech-Converter-with-amazone-polly.git (push)

# Show detailed info about the remote
git remote show origin


----------------------------------------------------------------
  SECTION 5 — REMOVE REMOTE
----------------------------------------------------------------

# Remove the remote named "origin"
git remote remove origin

# Verify it was removed (should return nothing)
git remote -v

# ⚠️  This only removes the remote link from your local config.
#     It does NOT delete anything on GitHub.

# Re-add it again after removing (if needed)
git remote add origin https://github.com/aamir490/Use-case--9-NovaMind-AI-Text-to-Speech-Converter-with-amazone-polly.git


----------------------------------------------------------------
  SECTION 6 — RENAME REMOTE (optional)
----------------------------------------------------------------

# Rename remote from "origin" to another name
git remote rename origin upstream

# Rename it back to origin
git remote rename upstream origin


----------------------------------------------------------------
  SECTION 7 — STAGE FILES
----------------------------------------------------------------

# Stage ALL files in the project
git add .

git add README.md
git add interview.md
git add github.txt
git add steps_to_do.md

# Stage all .py files
git add *.py

git add Load_On_S3_or_ELB/

# Check what is staged and what is not
git status

git diff --staged

----------------------------------------------------------------
  SECTION 9 — PUSH TO GITHUB
----------------------------------------------------------------

# Push to GitHub for the first time (sets upstream tracking)
git push -u origin main

# Push on subsequent commits
git push

# Push to a specific branch
git push origin main
git push origin dev

# Force push (⚠️ use with caution — overwrites remote history)
git push --force origin main

# Push and set upstream in one command
git push --set-upstream origin main


----------------------------------------------------------------
  SECTION 10 — BRANCH MANAGEMENT
----------------------------------------------------------------

# List all local branches
git branch

# List all branches including remote
git branch -a

# Create a new branch
git branch dev
git branch feature/add-cloudfront

# Switch to a branch
git checkout dev
git checkout main

# Create and switch in one command
git checkout -b feature/add-authentication

# Push a new branch to GitHub
git push -u origin feature/add-authentication

# Delete a local branch
git branch -d dev

# Delete a remote branch on GitHub
git push origin --delete dev


----------------------------------------------------------------
  SECTION 11 — PULL FROM GITHUB
----------------------------------------------------------------

# Fetch and merge latest changes from remote
git pull origin main

# Pull with rebase (cleaner history)
git pull --rebase origin main

# Fetch without merging (safe — just downloads)
git fetch origin


----------------------------------------------------------------
  SECTION 12 — VIEW HISTORY & LOGS
----------------------------------------------------------------

# Show full commit history
git log

# Show compact one-line log
git log --oneline

# Show graphical branch log
git log --oneline --graph --all

# Show last 5 commits
git log -5

# Show changes in a specific commit
git show <commit-hash>


----------------------------------------------------------------
  SECTION 13 — UNDO / RESET
----------------------------------------------------------------

# Unstage a file (keep changes in working directory)
git restore --staged README.md

# Discard changes in a file (⚠️ permanent)
git restore README.md

# Undo the last commit but keep changes staged
git reset --soft HEAD~1

# Undo the last commit and unstage changes
git reset --mixed HEAD~1

# ⚠️ Hard reset — undo commit AND discard all changes
git reset --hard HEAD~1

# Revert a specific commit (safe — creates a new undo commit)
git revert <commit-hash>
